    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const yesBtn = document.getElementById('yesBtn');
            const noBtn = document.getElementById('noBtn');
            const resultContainer = document.getElementById('resultContainer');
            const buttonsContainer = document.querySelector('.buttons-container');
            
            let clickCount = 0;
            const messages = [
                "Are you sure, June? 💕",
                "But it's Valentine's Day!",
                "Think of all the chocolate!",
                "Please reconsider? 🥺",
                "Just one more chance?",
                "Pretty please, June? 💘",
                "I brought virtual roses!",
                "You're breaking my heart!",
                "Final chance? 💝",
                "But I made you a card!",
                "For Ami? Please? 💗",
                "Be my Valentine?",
                "I'll keep trying!",
                "Okay, last try! 💓"
            ];
            
            // Function to create floating hearts
            function createHeart() {
                const heart = document.createElement('div');
                heart.classList.add('flower-animation');
                const hearts = ['💘', '💝', '💖', '💗', '💓', '💕', '💞', '❤️', '🧡', '💛', '💚', '💙', '💜', '🤎', '🖤', '🤍'];
                heart.innerHTML = hearts[Math.floor(Math.random() * hearts.length)];
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.fontSize = (Math.random() * 30 + 20) + 'px';
                heart.style.color = Math.random() > 0.5 ? '#ff4d6d' : '#b15eff';
                document.body.appendChild(heart);
                
                setTimeout(() => {
                    heart.remove();
                }, 6000);
            }
            
            // Function to create personalized Valentine hearts
            function createValentineHeart() {
                const heart = document.createElement('div');
                heart.classList.add('flower-animation');
                heart.innerHTML = Math.random() > 0.5 ? 'J❤️' : 'A❤️';
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.fontSize = (Math.random() * 35 + 25) + 'px';
                heart.style.color = Math.random() > 0.5 ? '#ff4d6d' : '#b15eff';
                heart.style.fontWeight = 'bold';
                document.body.appendChild(heart);
                
                setTimeout(() => {
                    heart.remove();
                }, 6000);
            }
            
            // Create some initial hearts
            for(let i = 0; i < 8; i++) {
                setTimeout(createHeart, i * 200);
            }
            
            // Make the "No" button move when hovered
            noBtn.addEventListener('mouseover', function() {
                moveNoButton();
            });
            
            noBtn.addEventListener('touchstart', function(e) {
                e.preventDefault();
                moveNoButton();
            });
            
            // Make the "No" button move when clicked
            noBtn.addEventListener('click', function(e) {
                e.preventDefault();
                moveNoButton();
                
                // Change the text after a few tries
                if (clickCount < messages.length) {
                    noBtn.innerHTML = `<i class="fas fa-times"></i> ${messages[clickCount]}`;
                    clickCount++;
                    
                    // Make button smaller as she tries more
                    const scale = Math.max(0.65, 1 - (clickCount * 0.025));
                    noBtn.style.transform = `scale(${scale})`;
                    
                    // Make button more transparent
                    noBtn.style.opacity = Math.max(0.5, 1 - (clickCount * 0.03));
                }
                
                // Create hearts when "No" is clicked
                for(let i = 0; i < 4; i++) {
                    setTimeout(createValentineHeart, i * 150);
                }
                
                // Sad animation on yes button
                yesBtn.style.animation = 'pulse 1s';
                setTimeout(() => {
                    yesBtn.style.animation = 'glow 2s infinite';
                }, 1000);
            });
            
            function moveNoButton() {
                const containerRect = buttonsContainer.getBoundingClientRect();
                const buttonRect = noBtn.getBoundingClientRect();
                
                // Calculate random position within container
                const maxX = containerRect.width - buttonRect.width - 20;
                const maxY = containerRect.height - buttonRect.height - 20;
                
                const randomX = Math.random() * maxX;
                const randomY = Math.random() * maxY;
                
                // Apply new position
                noBtn.style.left = randomX + 'px';
                noBtn.style.top = randomY + 'px';
                
                // Add a little rotation for fun
                const rotation = Math.random() * 30 - 15;
                noBtn.style.transform = `rotate(${rotation}deg) scale(${Math.max(0.65, 1 - (clickCount * 0.025))})`;
                
                // Create a heart at the old position
                createHeart();
            }
            
            // Handle "Yes" button click
            yesBtn.addEventListener('click', function() {
                // Show result container
                resultContainer.style.display = 'block';
                
                // Scroll to result
                resultContainer.scrollIntoView({ behavior: 'smooth' });
                
                // Create celebration hearts
                for(let i = 0; i < 40; i++) {
                    setTimeout(() => {
                        if (i % 3 === 0) {
                            createHeart();
                        } else if (i % 3 === 1) {
                            createValentineHeart();
                        } else {
                            const love = document.createElement('div');
                            love.classList.add('flower-animation');
                            love.innerHTML = '💘 YES! 💘';
                            love.style.left = Math.random() * 100 + 'vw';
                            love.style.fontSize = '28px';
                            love.style.color = '#ff4d6d';
                            love.style.fontWeight = 'bold';
                            document.body.appendChild(love);
                            setTimeout(() => love.remove(), 6000);
                        }
                    }, i * 120);
                }
                
                // Play celebration animation
                yesBtn.style.transform = 'scale(1.3)';
                yesBtn.innerHTML = '<i class="fas fa-heart"></i> Thank you, my Valentine! 💘';
                yesBtn.style.background = 'linear-gradient(to right, #ff4d6d, #ff8fa3)';
                setTimeout(() => {
                    yesBtn.style.transform = 'scale(1)';
                }, 600);
                
                // Change the page title
                document.title = "June said YES! 💘 | Valentine's Day";
                
                // Hide the no button
                noBtn.style.opacity = '0.3';
                noBtn.style.pointerEvents = 'none';
                noBtn.innerHTML = '<i class="fas fa-heart"></i> She said yes! 💘';
                
                // Create floating "JUNE & AMI" text
                for(let i = 0; i < 15; i++) {
                    setTimeout(() => {
                        const text = document.createElement('div');
                        text.classList.add('flower-animation');
                        const messages = ['JUNE 💘 AMI', 'VALENTINES', 'LOVE', '💞', 'FOREVER', '❤️', 'BE MINE', 'J❤️A'];
                        text.innerHTML = messages[Math.floor(Math.random() * messages.length)];
                        text.style.left = Math.random() * 100 + 'vw';
                        text.style.fontSize = '30px';
                        text.style.fontWeight = 'bold';
                        text.style.color = ['#ff4d6d', '#b15eff', '#ff8fa3'][Math.floor(Math.random() * 3)];
                        text.style.textShadow = '2px 2px 4px rgba(0,0,0,0.2)';
                        document.body.appendChild(text);
                        
                        setTimeout(() => {
                            text.remove();
                        }, 6000);
                    }, i * 200);
                }
            });
            
            // Add some random heart animations periodically
            setInterval(() => {
                if (Math.random() > 0.6) createHeart();
                if (Math.random() > 0.7) createValentineHeart();
            }, 2500);
        });
    </script>
